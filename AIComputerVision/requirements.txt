# Pinned runtime dependencies for the AIComputerVision training/export pipeline.
#
#   pip install -r AIComputerVision/requirements.txt
#
# The desktop application does not need these packages at runtime - it only
# loads the exported ONNX files from AIComputerVision/Models/. They are needed
# to (re)train models via TrainModel.py.
#
# Versions below match the environment the current models were produced with.
# ``torch`` is a range, not an exact pin, because the CUDA and CPU builds share
# the same version number and are selected by the index used to install them:
# https://pytorch.org/get-started/locally/ . The range keeps a run on a
# supported ABI without silently accepting a major framework jump.
torch>=2.7,<3
ultralytics==8.4.41
PyYAML==6.0.2
onnx==1.21.0
onnxslim==0.1.91
# Post-export compatibility validation (the same engine the Qt runtime uses).
# Exactly the version CMakeLists.txt pins (ONNXRUNTIME_VERSION): the export
# contract is validated against the engine the desktop application ships, so
# "the artifact loads" has to mean "it loads under the runtime that ships".
# A range would let a newer engine validate an artifact the shipped runtime
# cannot load. The drift between this file and CMakeLists.txt is a red test
# (tests/test_toolchain_contract.py), not a reviewer's memory.
onnxruntime==1.20.1

# Optional - only required when training with ``--tensorboard``.
tensorboard==2.15.2
